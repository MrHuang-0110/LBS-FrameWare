#include "telemetry.hpp"
#include "device_controller.hpp"

constexpr uint32_t SAMPLE_INTERVAL_MS = 500;

void setup() {
  DeviceController::begin();
  Telemetry::setSampleInterval(SAMPLE_INTERVAL_MS);
  Telemetry::onThresholdExceeded([](SensorType type, float value) {
    Logger::warn("Threshold exceeded: {} = {}", type, value);
    DeviceController::setSafeMode(true);
  });
}

void loop() {
  Telemetry::sample();
  DeviceController::process();
  delay(10);
}
